```js
// ① 造一个**真 game**（JS 引擎状态）—— 注意 team 必须是 3 个 SPECIES 的 id
import {createGame, SPECIES} from './src/game/engine.js';        // engine.js:121
const ids = SPECIES.slice(0, 3).map((p) => p.id);                // 例如 ['fox','turtle','deer']
const game = createGame(17, ids);
//   ⚠ 约束（engine.js:122 逐字）：team 必须 **恰好 3 只、互不相同、且都是 SPECIES 里的 id**

// ② 装配 context —— **产品前端就是这么造的**（src/client/roco.js:4961）
const ctx = {mode: game.mode ?? 'battle', battle: game};

// ③ 真实执行（src/coach/toolbox.js 导出）
import {executeTool} from './src/coach/toolbox.js';
const receipt = await executeTool('read_state', {}, ctx, '现在场上是什么情况？');
```

### 10.3 ⚠ 我踩过的两个错（**别重踩**）

```
① 手搓 `{view: <Python 服务回执>}`      ⇒ 返回**空壳**（turn:null、legal 空）✗（键名不对）
② 手搓 `{battle: <Python 的 view>}`     ⇒ **TypeError: … reading 'pets'** at legalActions ✗
⇒ **`context.battle` 是 JS 引擎的 `game`，不是 Python 的 `view`** ✗
```
